# Install AegisPay Agent SDK
import requests

def execute_agent_payment(agent_id, intent, merchant, amount):
    """Autonomously purchases via AegisPay PayPal Enclave"""
    payload = {
        "agent_id": agent_id,
        "intent": intent,
        "merchant": merchant,
        "amount": amount,
        "currency": "USD"
    }
    resp = requests.post(
        "http://localhost:8080/api/agent/transact",
        json=payload
    )
    result = resp.json()
    if result.get("success"):
        order_id = result["transaction"]["paypal_order_id"]
        print(f"Settled via PayPal: {order_id}")
    else:
        print(f"Blocked by Policy: {result.get('evaluation')}")
    return result
